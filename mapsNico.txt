const MapaModal = ({ show, onHide }) => {
  const [direccion, setDireccion] = useState("")
  const [localidad, setLocalidad] = useState("")
  const [provincia, setProvincia] = useState("")
  const [pais, setPais] = useState("")
  const [urlMapa, setUrlMapa] = useState("")

  const buscarEnMapa = () => {
    const partes = [direccion, localidad, provincia, pais].filter(p => p.trim() !== '')
    const busqueda = partes.join(', ')
    
    if (!busqueda) {
      alert('Por favor, completa al menos un campo para buscar')
      return
    }
    
    // Usar URL estándar de Google Maps (funciona sin API Key)
    setUrlMapa(
      https://www.google.com/maps?q=${encodeURIComponent(busqueda)}&output=embed&t=k
    )
  }

  const limpiarCampos = () => {
    setDireccion('')
    setLocalidad('')
    setProvincia('')
    setPais('')
    setUrlMapa('')
  }

  const usarEjemplo = (ejemplo) => {
    setDireccion(ejemplo.direccion || '')
    setLocalidad(ejemplo.localidad || '')
    setProvincia(ejemplo.provincia || '')
    setPais(ejemplo.pais || '')
    
    const partes = [ejemplo.direccion, ejemplo.localidad, ejemplo.provincia, ejemplo.pais].filter(p => p && p.trim() !== '')
    const busqueda = partes.join(', ')
    
    if (busqueda) {
      setUrlMapa(
        https://www.google.com/maps?q=${encodeURIComponent(busqueda)}&output=embed&t=k
      )
    }
  }

  return (
    <Modal 
      show={show} 
      onHide={onHide} 
      size="xl" 
      centered
      className="mapa-modal"
    >
      <Modal.Header closeButton className="modal-header-custom">
        <Modal.Title className="modal-title-custom">
          <span className="title-icon">📍</span>
          Buscar ubicación en Google Maps
        </Modal.Title>
      </Modal.Header>

      <Modal.Body className="modal-body-custom">
        <Row className="g-0 h-100">
          <Col md={3} className="form-column">
            <div className="form-container">
              <h5 className="form-title">Datos de ubicación</h5>
              <p className="form-subtitle">Completa los campos para buscar</p>
              
              <Form>
                <Form.Group className="mb-3">
                  <Form.Label className="field-label-custom">
                    <span className="label-icon">🏠</span> Dirección
                  </Form.Label>
                  <Form.Control
                    type="text"
                    value={direccion}
                    onChange={(e) => setDireccion(e.target.value)}
                    placeholder="Ej: Av. Corrientes 1234"
                    className="input-custom"
                  />
                </Form.Group>

                <Form.Group className="mb-3">
                  <Form.Label className="field-label-custom">
                    <span className="label-icon">🏙️</span> Localidad
                  </Form.Label>
                  <Form.Control
                    type="text"
                    value={localidad}
                    onChange={(e) => setLocalidad(e.target.value)}
                    placeholder="Ej: Buenos Aires"
                    className="input-custom"
                  />
                </Form.Group>

                <Form.Group className="mb-3">
                  <Form.Label className="field-label-custom">
                    <span className="label-icon">🗺️</span> Provincia
                  </Form.Label>
                  <Form.Control
                    type="text"
                    value={provincia}
                    onChange={(e) => setProvincia(e.target.value)}
                    placeholder="Ej: Buenos Aires"
                    className="input-custom"
                  />
 </Form.Group>

                <Form.Group className="mb-4">
                  <Form.Label className="field-label-custom">
                    <span className="label-icon">🌍</span> País
                  </Form.Label>
                  <Form.Control
                    type="text"
                    value={pais}
                    onChange={(e) => setPais(e.target.value)}
                    placeholder="Ej: Argentina"
                    className="input-custom"
                  />
                </Form.Group>

                <div className="button-group">
                  <Button 
                    variant="primary" 
                    onClick={buscarEnMapa}
                    className="btn-buscar"
                  >
                    🔍 Buscar en mapa
                  </Button>
                  <Button 
                    variant="outline-secondary" 
                    onClick={limpiarCampos}
                    className="btn-limpiar"
                  >
                    🗑️ Limpiar
                  </Button>
                </div>

                <hr className="divider" />

                <div className="ejemplos-container">
                  <p className="ejemplos-title">📌 Ejemplos rápidos</p>
                  <div className="ejemplos-grid">
                    <Button 
                      variant="outline-primary" 
                      size="sm"
                      className="btn-ejemplo"
                      onClick={() => usarEjemplo({
                        direccion: 'Av. Corrientes 1234',
                        localidad: 'Buenos Aires',
                        provincia: 'Buenos Aires',
                        pais: 'Argentina'
                      })}
                    >
                      Ejemplo 1
                    </Button>
                    <Button 
                      variant="outline-primary" 
                      size="sm"
                      className="btn-ejemplo"
                      onClick={() => usarEjemplo({
                        direccion: 'Calle 52 418',
                        localidad: 'La Plata',
                        provincia: 'Buenos Aires',
                        pais: 'Argentina'
                      })}
                    >
                      Ejemplo 2
                    </Button>
                    <Button

                      size="sm"
                      className="btn-ejemplo"
                      onClick={() => usarEjemplo({
                        direccion: 'Av. Mayo 525',

                        provincia: 'Buenos Aires',
                        pais: 'Argentina'
                      })}
                    >
                      Ejemplo 3
                    </Button>
                  </div>
                </div>
              </Form>
            </div>
          </Col>

          <Col md={9} className="map-column">
            <div className="map-container">
              {urlMapa ? (
 <iframe
                  title="Google Maps"
                  src={urlMapa}
                  width="100%"
                  height="100%"
                  style={{ 
                    border: 0, 
                    borderRadius: '12px',
                    boxShadow: '0 4px 20px rgba(0,0,0,0.1)'
                  }}
                  allowFullScreen
                  loading="lazy"
                />
              ) : (
                <div className="map-placeholder">
                  <div className="placeholder-content">
                    <div className="placeholder-icon">🗺️</div>
                    <h4>Ingresa una ubicación</h4>
                    <p className="text-muted">
                      Completa los campos a la izquierda<br />
                      y presiona "Buscar en mapa"
                    </p>
                    <div className="placeholder-examples">
                      <small>Ejemplos: </small>
                      <span className="badge-example">Av. Corrientes 1234, Buenos Aires</span>
                      <span className="badge-example">Calle 52 418, La Plata</span>
                    </div>
                  </div>
                </div>
              )}
            </div>
          </Col>
        </Row>
      </Modal.Body>
    </Modal>
  )
}

function App() {
  const [showMapaModal, setShowMapaModal] = useState(false)

  return (
    <div className="app-container">
      <header className="app-header">
        <h1>📍 Buscador de Ubicaciones</h1>
        <p>Encuentra cualquier lugar en Google Maps Satelital</p>
        <Button 
          variant="primary" 
          size="lg"
          onClick={() => setShowMapaModal(true)}
          className="btn-abrir-mapa"
        >
          🗺️ Abrir buscador de mapa
        </Button>
      </header>

      <MapaModal 
        show={showMapaModal}
        onHide={() => setShowMapaModal(false)}
      />
    </div>
  )
}

export default App
